
Recipes: a session's files, for whoever would rather script than ask.
A session is a fux pane: a folder of plain files on the server, in the folder `koh serve`
printed (`sessions dir:`: `$XDG_RUNTIME_DIR/koh/fux/koh-sessions`, or with no
`XDG_RUNTIME_DIR` the temporary directory's `koh-UID/fux/koh-sessions`), and named
as `koh sessions` names it. `koh sessions`, `koh output` and `koh wait --session` read
those files there and no more; the script below is nearly the same as sh, for `koh exec`
or for a script of your own on the server:

  koh help-agents | sed -n '/^#!/,$p' | koh rsh SERVER 'cat > sessions.sh'
  koh exec SERVER -- sh sessions.sh list DIR          DIR: the folder `koh serve` printed
  koh exec SERVER -- sh sessions.sh wait DIR/NAME 60  and `screen`, `last` likewise

What is typed goes into a session's `in`, a FIFO: `koh exec SERVER --sh 'printf "make\n" >
DIR/NAME/in'`. Ask a wait before you type what it is to wait for (`koh wait … &`, then type,
then `wait $!`): it is for the first command that ends after it was asked, and one that has
ended by then is not that. A shell must mark its prompts for `last` and `wait` (OSC 133); for
bash from 4.4:
  PS1='\[\e]133;A\a\]'$PS1'\[\e]133;B\a\]'; PS0='\e]133;C\a'
  PROMPT_COMMAND='printf "\e]133;D;%s\a" "$?"'
fux's README, "A pane's files", says what every file is.

#!/bin/sh
# sessions.sh list DIR | screen PANE | last PANE | wait PANE [SECONDS]
set -u
# Whether a holder keeps the pane, by its number: one that was given to another process
# since misleads. Only the lock on the folder is exact, which `koh` asks (and, on Linux,
# `flock -n PANE true`: it fails while the holder lives).
alive() { [ -p "$1/in" ] && kill -0 "$(sed -n '1s/^holder //p' "$1/status" 2>/dev/null)" 2>/dev/null; }

list() { # NAME <tab> alive|ended <tab> STATE <tab> COLSxROWS <tab> busy|idle <tab> last exit or - <tab> TITLE
  for p in "$1"/*/; do
    p=${p%/}
    [ -f "$p/status" ] || continue
    if alive "$p"; then a=alive; else a=ended; fi
    cols=0 rows=0; { read -r cols rows < "$p/size"; } 2> /dev/null   # (a pane that has only begun has none yet)
    # A command runs while the holder lives and the last mark is `output`: after `command` the shell waits at its prompt.
    work=$(awk -v a=$a '$3 == "end" { code = $4 } { last = $3 }
      END { if (code == "") code = "-"; printf "%s\t%s", (a == "alive" && last == "output" ? "busy" : "idle"), code }' "$p/marks" 2> /dev/null) || work=$(printf 'idle\t-')
    printf '%s\t%s\t%s\t%sx%s\t%s\t%s\n' "${p##*/}" "$a" "$(cat "$p/state" 2>/dev/null)" "$cols" "$rows" "$work" "$(cat "$p/title" 2>/dev/null)"
  done
}

# text PANE FILE: the pane's text into FILE (history.1, history, the main screen with the rows that
# go on joined), and prints what to add to a line number of `marks` to have that line's number in FILE.
text() {
  { read -r top; read -r wraps; } < "$1/main/lines"
  open=$(tail -c 1 "$1/history" 2> /dev/null)
  case " $wraps " in *" 0 "*) open= ;; esac
  { cat "$1/history.1" "$1/history" 2> /dev/null; [ -z "$open" ] || echo; } > "$2"
  before=$(($(wc -l < "$2")))
  awk -v wraps="$wraps" 'BEGIN { split(wraps, w, " "); for (i in w) joined[w[i] + 1] = 1 } { row[NR] = $0; if ($0 != "") last = NR }
    END { for (i = 1; i <= last; i++) printf "%s%s", (i > 1 && !joined[i] ? "\n" : ""), row[i]; if (last) print "" }' "$1/main/screen" >> "$2"
  echo $((before - top + 1))
}

last() { # what the last command that has ended printed, or with N the first whose `end` is numbered above N: from its `output` mark to its `end` mark
  f=$(mktemp) m=$(mktemp) tries=0
  # Of one round of the holder's writing: `generation` the same even number before and after
  # (ten tries: a pane that never stops printing, or whose holder was killed as it wrote, is taken as it is).
  while g=$(cat "$1/generation"); add=$(text "$1" "$f"); cat "$1/marks.dropped" "$1/marks" > "$m"
    { [ $((${g:-1} % 2)) -eq 1 ] || [ "$g" != "$(cat "$1/generation")" ]; } && [ $tries -lt 10 ]; do tries=$((tries + 1)); sleep 0.02; done
  # (Cut at the marks' bytes, as `koh output --last` and fux's key `C-b C` cut, with a line break after the last line.)
  LC_ALL=C awk -v add="$add" -v n="${2:-}" 'NR == FNR { if (FNR == 1) d = $1; else if (!done) { if ($3 == "output") { o = $1; oa = $2 }
      if ($3 == "end" && o != "") { from = o; fa = oa; to = $1; ta = $2; o = "" }; done = n != "" && $3 == "end" && d + FNR - 1 > n }; next }
    from != "" && FNR >= from + add && FNR <= to + add { l = FNR == to + add ? substr($0, 1, ta) : $0
      if (FNR == from + add) l = substr(l, fa + 1); if (FNR < to + add || l != "") print l }' "$m" "$f"
  rm -f "$f" "$m"
}

# ends PANE: how many marks the pane has dropped, and then each `end` it lists as NUMBER CODE: line n
# of `marks` is the mark numbered `marks.dropped` + n, for as long as it is listed. Of one round
# (`generation` the same even number before and after), or nothing: look again.
ends() {
  g=$(cat "$1/generation" 2> /dev/null) && [ $((${g:-1} % 2)) -eq 0 ] || return
  e=$(awk 'NR == 1 { n = $1; print n; next } $3 == "end" { print n + FNR, $4 + 0 }' "$1/marks.dropped" "$1/marks" 2> /dev/null) || return
  [ "$g" = "$(cat "$1/generation" 2> /dev/null)" ] && printf '%s\n' "$e"
}

await() { # until a command has ended since this was asked (its exit code is this one's), or the session has; 124 after SECONDS
  # Asked: the number of the last `end`, or with none of the last mark that was dropped.
  # (Of a session that ends as this is asked there may be no reading: then no `end` is waited for.)
  until asked=$(ends "$1") || ! alive "$1"; do sleep 0.02; done
  asked=$(printf '%s\n' "$asked" | awk '{ e = $1; read = ($0 != "") } END { if (read) print e + 0 }') began=$(date +%s)
  while :; do
    # The first `end` of a higher number is the command, and how many marks came and went unread.
    now=; [ -z "$asked" ] || now=$(ends "$1" | awk -v e="$asked" 'NR == 1 { gone = $1 > e ? $1 - e : 0; next } $1 > e { print $2, gone; exit }')
    [ -z "$now" ] || {
      [ "${now#* }" = 0 ] || echo "the oldest end still kept, not known to be the first: ${now#* } marks came and went unread" >&2
      last "$1" "$asked"; exit "${now% *}"; }
    alive "$1" || { cat "$1/shown/screen"; s=$(cat "$1/state")
      case $s in "exited "*) exit "${s#exited }" ;; "killed "*) exit $((128 + ${s#killed })) ;; *) exit 1 ;; esac; }
    [ -z "${2:-}" ] || [ $(($(date +%s) - began)) -lt "$2" ] || { cat "$1/shown/screen"; exit 124; }
    sleep 0.1
  done
}

case ${1:-} in
  list) list "$2" ;;
  screen) cat "$2/shown/screen" ;;
  last) last "$2" ;;
  wait) await "$2" "${3:-}" ;;
  *) echo "sessions.sh list DIR | screen PANE | last PANE | wait PANE [SECONDS]" >&2; exit 2 ;;
esac
